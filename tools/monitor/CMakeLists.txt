cmake_minimum_required(VERSION 3.16)
project(BHumanMonitor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SFML 2.5 REQUIRED COMPONENTS graphics window system)
find_package(jsoncpp CONFIG REQUIRED)
add_library(monitor_model Model.cpp)
target_link_libraries(monitor_model PUBLIC jsoncpp_lib)
target_include_directories(monitor_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(bhuman_monitor Main.cpp)
target_link_libraries(bhuman_monitor PRIVATE monitor_model sfml-graphics sfml-window sfml-system)
target_compile_options(bhuman_monitor PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(monitor_model_test ModelTest.cpp)
target_link_libraries(monitor_model_test PRIVATE monitor_model)
add_test(NAME monitor_model COMMAND monitor_model_test)
